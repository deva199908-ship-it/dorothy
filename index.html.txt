<script>
function escapeNo(){

  let no = document.getElementById("noBtn");
  let yes = document.getElementById("yesBtn");

  // Random position inside screen (avoid edges)
  let maxX = window.innerWidth - 200;
  let maxY = window.innerHeight - 100;

  no.style.position = "fixed";
  no.style.left = Math.random() * maxX + "px";
  no.style.top = Math.random() * maxY + "px";

  // Increase YES size
  let size = parseInt(window.getComputedStyle(yes).fontSize);
  yes.style.fontSize = (size + 8) + "px";

  // Add glowing effect
  yes.style.boxShadow = "0 0 30px #ff0066";

  // Add more hearts
  let count = yes.getAttribute("data-hearts") || 1;
  count = parseInt(count) + 1;
  yes.setAttribute("data-hearts", count);

  yes.innerHTML = "YES " + "💖".repeat(count);
}

/* Floating Hearts */
setInterval(()=>{
  const heart=document.createElement("div");
  heart.classList.add("heart");
  heart.innerHTML="💖";
  heart.style.left=Math.random()*100+"vw";
  heart.style.animationDuration=Math.random()*3+3+"s";
  heart.style.fontSize="20px";
  document.body.appendChild(heart);
  setTimeout(()=>{heart.remove()},6000);
},300);

/* YES grows when NO hover */
function growYes(){
  let yes = document.getElementById("yesBtn");

  // Increase size
  let size = parseInt(window.getComputedStyle(yes).fontSize);
  yes.style.fontSize = (size + 6) + "px";

  // Change text with more hearts
  let currentHearts = yes.getAttribute("data-hearts") || 1;
  currentHearts = parseInt(currentHearts) + 1;

  yes.setAttribute("data-hearts", currentHearts);

  let hearts = "💖".repeat(currentHearts);
  yes.innerHTML = "YES " + hearts;
}


/* Start Love Mode */
function startLove(){

  // Hide NO button completely
  document.getElementById("noBtn").style.display = "none";

  // Show main section
  document.getElementById("main").classList.remove("hidden");

  // Play music
  document.getElementById("bgMusic").play();

  // Fireworks
  for(let i=0;i<30;i++){
    let fire=document.createElement("div");
    fire.classList.add("firework");
    fire.style.left=Math.random()*100+"vw";
    fire.style.top="80vh";
    fire.style.background="hsl("+Math.random()*360+",100%,60%)";
    document.body.appendChild(fire);
    setTimeout(()=>{fire.remove()},1000);
  }
}
